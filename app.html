// ১. স্টেট ভেরিয়েবলস (ওয়ালেটের অবস্থা ট্র্যাকিং) let userAptosAddress = null; let isConnected = false; // ২. Petra প্রোভাইডার চেক করার ফাংশন function getPetraProvider() { if ('aptos' in window) return window.aptos; if ('petra' in window) return window.petra; return null; } // ৩. মেইন হ্যান্ডলার (বাটনে ক্লিক করলে যা হবে) function handleWalletAction() { if (!isConnected) { connectWallet(); } else { disconnectWallet(); } } // ৪. ওয়ালেট কানেক্ট করার লজিক async function connectWallet() { const provider = getPetraProvider(); if (!provider) { alert("Petra Aptos Wallet খুঁজে পাওয়া যায়নি! দয়া করে এক্সটেনশনটি ইনস্টল করুন।"); window.open("https://petra.app/", "_blank"); return; } try { // Petra ওয়ালেটে কানেকশন রিকোয়েস্ট পাঠানো const response = await provider.connect(); // অ্যাড্রেস ও স্টেট সেভ করা userAptosAddress = response.address; isConnected = true; // UI আপডেট করা setupConnectedUI(userAptosAddress); alert("ওয়ালেট সফলভাবে কানেক্ট হয়েছে! 🎉"); } catch (error) { console.error("Connection rejected:", error); alert("কানেকশন রিকোয়েস্ট বাতিল করা হয়েছে।"); } } // ৫. ওয়ালেট ডিসকানেক্ট করার লজিক async function disconnectWallet() { const provider = getPetraProvider(); if (!provider) return; try { await provider.disconnect(); // স্টেট ক্লিয়ার করা userAptosAddress = null; isConnected = false; // UI রিসেট করা resetWalletUI(); alert("ওয়ালেট ডিসকানেক্ট করা হয়েছে।"); } catch (error) { console.error("Disconnection failed:", error); } } // ৬. কানেক্টেড UI এবং হোভার ইফেক্ট (ভিডিওর মতো এক্সাক্ট লজিক) function setupConnectedUI(address) { const walletBtn = document.getElementById("walletBtn"); if (!walletBtn) return; // অ্যাড্রেস ছোট করা (যেমন: 0x4f1f...3eb5) const shortAddr = address.substring(0, 6) + "..." + address.substring(address.length - 4); walletBtn.innerText = shortAddr; walletBtn.classList.add("connected"); // মাউস উপরে আনলে "Disconnect X" দেখাবে walletBtn.onmouseover = function() { this.innerText = "Disconnect X"; this.classList.add("disconnect-mode"); }; // মাউস সরিয়ে নিলে আবার অ্যাড্রেস দেখাবে walletBtn.onmouseout = function() { this.innerText = shortAddr; this.classList.remove("disconnect-mode"); }; } // ৭. ডিসকানেক্টের পর UI আগের অবস্থায় নেওয়া function resetWalletUI() { const walletBtn = document.getElementById("walletBtn"); if (!walletBtn) return; walletBtn.innerText = "Connect Wallet"; walletBtn.className = "btn-wallet"; // ডিফল্ট ক্লাসে ফেরত নেওয়া // হোভার ইভেন্টগুলো মুছে ফেলা walletBtn.onmouseover = null; walletBtn.onmouseout = null; }
